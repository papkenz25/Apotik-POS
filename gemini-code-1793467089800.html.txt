    <!-- SCRIPT LOGIKA KASIR -->
    <script>
        // Simulasi Data Obat (Bisa disesuaikan atau ditarik otomatis dari Sheets)
        let katalogObat = [
            { id: 1, nama: "Amoxicillin", harga: 15000, stok: 3 },
            { id: 2, nama: "CTM 4mg", harga: 3000, stok: 5 },
            { id: 3, nama: "Paracetamol", harga: 5000, stok: 20 },
            { id: 4, nama: "Vitamin C", harga: 10000, stok: 15 },
            { id: 5, nama: "Ibuprofen", harga: 8000, stok: 8 }
        ];

        let keranjang = [];
        let webhookUrl = localStorage.getItem('n8n_webhook_url') || '';

        // Tampilkan daftar obat saat halaman dimuat
        window.onload = function() {
            tampilkanKatalog(katalogObat);
        };

        function tampilkanKatalog(data) {
            const container = document.getElementById('daftarObat');
            container.innerHTML = '';

            if (data.length === 0) {
                container.innerHTML = `<p class="text-slate-400 text-center col-span-2 py-8 text-sm">Obat tidak ditemukan.</p>`;
                return;
            }

            data.forEach(item => {
                let statusStok = item.stok < 10 
                    ? `<span class="text-xs bg-red-100 text-red-600 px-2 py-0.5 rounded-full font-medium">Stok Kritis: ${item.stok}</span>` 
                    : `<span class="text-xs bg-emerald-100 text-emerald-600 px-2 py-0.5 rounded-full font-medium">Stok: ${item.stok}</span>`;

                container.innerHTML += `
                    <div class="border border-slate-100 bg-slate-50/50 hover:bg-white hover:border-emerald-200 transition p-4 rounded-xl flex flex-col justify-between">
                        <div>
                            <div class="flex justify-between items-start mb-1">
                                <h3 class="font-bold text-slate-700 text-sm">${item.nama}</h3>
                                ${statusStok}
                            </div>
                            <p class="text-emerald-600 font-semibold text-sm mb-3">Rp ${item.harga.toLocaleString('id-ID')}</p>
                        </div>
                        <button onclick="tambahKeKeranjang(${item.id})" class="w-full bg-slate-900 hover:bg-emerald-600 text-white py-2 rounded-lg text-xs font-medium transition flex items-center justify-center space-x-1">
                            <i class="fa-solid fa-cart-plus"></i>
                            <span>Pilih Obat</span>
                        </button>
                    </div>
                `;
            });
        }

        function filterObat() {
            let keyword = document.getElementById('inputCari').value.toLowerCase();
            let hasil = katalogObat.filter(o => o.nama.toLowerCase().includes(keyword));
            tampilkanKatalog(hasil);
        }

        function tambahKeKeranjang(id) {
            let obat = katalogObat.find(o => o.id === id);
            let itemDiKeranjang = keranjang.find(i => i.id === id);

            if (itemDiKeranjang) {
                if (itemDiKeranjang.jumlah < obat.stok) {
                    itemDiKeranjang.jumlah++;
                } else {
                    alert("Jumlah melebihi sisa stok fisik!");
                }
            } else {
                if (obat.stok > 0) {
                    keranjang.push({ ...obat, jumlah: 1 });
                } else {
                    alert("Stok obat ini kosong!");
                }
            }
            renderKeranjang();
        }

        function ubahJumlah(id, delta) {
            let item = keranjang.find(i => i.id === id);
            let obat = katalogObat.find(o => o.id === id);

            if (item) {
                item.jumlah += delta;
                if (item.jumlah > obat.stok) {
                    alert("Jumlah melebihi sisa stok fisik!");
                    item.jumlah = obat.stok;
                }
                if (item.jumlah <= 0) {
                    keranjang = keranjang.filter(i => i.id !== id);
                }
            }
            renderKeranjang();
        }

        function renderKeranjang() {
            let container = document.getElementById('itemKeranjang');
            container.innerHTML = '';

            if (keranjang.length === 0) {
                container.innerHTML = `<p class="text-slate-400 text-center py-6 text-sm">Belum ada obat dipilih.</p>`;
                document.getElementById('labelTotal').innerText = 'Rp 0';
                document.getElementById('btnProses').disabled = true;
                return;
            }

            let totalBelanja = 0;
            keranjang.forEach(item => {
                let subtotal = item.harga * item.jumlah;
                totalBelanja += subtotal;

                container.innerHTML += `
                    py-3 flex justify-between items-center text-sm">
                        <div>
                            <p class="font-bold text-slate-700">${item.nama}</p>
                            <p class="text-xs text-slate-400">Rp ${item.harga.toLocaleString('id-ID')} x ${item.jumlah}</p>
                        </div>
                        <div class="flex items-center space-x-2">
                            <button onclick="ubahJumlah(${item.id}, -1)" class="w-7 h-7 bg-slate-200 hover:bg-slate-300 rounded-lg text-slate-600 font-bold">-</button>
                            <span class="text-sm font-bold w-5 text-center">${item.jumlah}</span>
                            <button onclick="ubahJumlah(${item.id}, 1)" class="w-7 h-7 bg-slate-200 hover:bg-slate-300 rounded-lg text-slate-600 font-bold">+</button>
                        </div>
                    </div>
                `;
            });

            document.getElementById('labelTotal').innerText = `Rp ${totalBelanja.toLocaleString('id-ID')}`;
            document.getElementById('btnProses').disabled = false;
            hitungKembalian();
        }

        function hitungKembalian() {
            let total = keranjang.reduce((sum, item) => sum + (item.harga * item.jumlah), 0);
            let bayar = parseFloat(document.getElementById('inputBayar').value) || 0;
            let kembalian = bayar - total;

            let labelKembalian = document.getElementById('labelKembalian');
            if (kembalian >= 0) {
                labelKembalian.innerText = `Rp ${kembalian.toLocaleString('id-ID')}`;
                labelKembalian.className = "font-bold text-emerald-600";
            } else {
                labelKembalian.innerText = `Kurang Rp ${Math.abs(kembalian).toLocaleString('id-ID')}`;
                labelKembalian.className = "font-bold text-red-500";
            }
        }

        function bukaPengaturan() {
            document.getElementById('inputWebhookUrl').value = webhookUrl;
            document.getElementById('modalPengaturan').classList.remove('hidden');
        }

        function tutupPengaturan() {
            document.getElementById('modalPengaturan').classList.add('hidden');
        }

        function simpanWebhook() {
            webhookUrl = document.getElementById('inputWebhookUrl').value.trim();
            localStorage.setItem('n8n_webhook_url', webhookUrl);
            tutupPengaturan();
            alert("URL Webhook n8n berhasil disimpan!");
        }

        async function prosesTransaksi() {
            if (!webhookUrl) {
                alert("Harap masukkan URL Webhook n8n terlebih dahulu melalui tombol Pengaturan di kanan atas!");
                bukaPengaturan();
                return;
            }

            let total = keranjang.reduce((sum, item) => sum + (item.harga * item.jumlah), 0);
            let bayar = parseFloat(document.getElementById('inputBayar').value) || 0;

            if (bayar < total) {
                alert("Uang tunai pembeli masih kurang!");
                return;
            }

            let dataTransaksi = {
                waktu: new Date().toISOString(),
                items: keranjang,
                totalBelanja: total,
                uangBayar: bayar,
                uangKembalian: bayar - total
            };

            try {
                let response = await fetch(webhookUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(dataTransaksi)
                });

                if (response.ok) {
                    alert("Transaksi Berhasil Diproses & Data Dikirim ke n8n!");
                    keranjang = [];
                    document.getElementById('inputBayar').value = '';
                    renderKeranjang();
                } else {
                    alert("Gagal mengirim data ke n8n. Periksa kembali URL Webhook.");
                }
            } catch (error) {
                console.error(error);
                alert("Terjadi kesalahan jaringan saat menghubungi n8n.");
            }
        }
    </script>
